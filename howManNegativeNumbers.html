static int howManyNegativeNumbers(int M[][])
{
// initialize result
int count = 0;
int comaprisonsCounter = 0;
// Start with top right corner
int i = 0;
int j = M[0].length - 1;
// Follow the path shown using
// arrows above
while( j >= 0 && i < M.length )
{
comaprisonsCounter++;
if( M[i][j] < 0 )
{
// j is the index of the
// last negative number
// in this row. So there
// must be ( j+1 )
count += j + 1;
// negative numbers in
// this row.
i += 1;
}
// move to the left and see
// if we can find a negative
// number there
else
j -= 1;
}
System.out.println("howManyNegativeNumbers : " + comaprisonsCounter);
return count;
}
